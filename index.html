Connect via Google Apps Script
The dashboard automatically creates one tab for every tab in your Google Sheet, whatever the tab names are. Each tab needs columns: A S.No · B Source Content · C Posted Content · D Views · E Revenue ($)
1In your Google Sheet → Extensions → Apps Script → delete ALL existing code → paste this:
function doGet(e) { const cache = CacheService.getScriptCache(); const forceRefresh = e && e.parameter && e.parameter.refresh === '1'; const cached = forceRefresh ? null : cache.get('sheet_data'); if (cached) { return ContentService.createTextOutput(cached).setMimeType(ContentService.MimeType.JSON); } const ss = SpreadsheetApp.getActiveSpreadsheet(); const out = {}; ss.getSheets().forEach(function(sheet) { const name = sheet.getName(); const lastRow = sheet.getLastRow(); if (lastRow < 1) { out[name] = []; return; } // Only read columns A-E (S.No, Source, Posted, Views, Revenue) const numCols = Math.min(sheet.getLastColumn(), 5); if (numCols < 1) { out[name] = []; return; } const values = sheet.getRange(1, 1, lastRow, numCols).getValues(); // Rich text (for hyperlinks) is only needed on columns B and C — // fetching it for every column is what made this slow before. const linkColStart = 2; // column B const linkColCount = Math.max(0, Math.min(numCols, 3) - 1); // covers B and C only const rich = linkColCount > 0 ? sheet.getRange(1, linkColStart, lastRow, linkColCount).getRichTextValues() : []; const rows = []; for (var r = 0; r < lastRow; r++) { var row = []; for (var c = 0; c < numCols; c++) { var v = String(values[r][c]); if (c === 1 || c === 2) { // column B or C try { var lnk = rich[r][c - 1].getLinkUrl(); if (lnk) v = lnk; } catch (ex) {} } row.push(v); } rows.push(row); } out[name] = rows; }); const json = JSON.stringify(out); cache.put('sheet_data', json, 21600); // cache for 6 hours return ContentService .createTextOutput(json) .setMimeType(ContentService.MimeType.JSON); }
